TASK 6: Business Rules
----------------------------------------------------------------------

--- Rule 1: Serial number format (XXX-NNN) ---
>> curl -s -X POST http://localhost:3000/equipment -H "Content-Type: application/json" -d '{"equipment":{"name":"Bad Serial","serial_number":"bad-01","status":"available","category_id":3}}'
{
  "errors": [
    "Serial number must look like ABC-123"
  ]
}

--- Rule 2: Maintenance date cannot be in the future ---
>> curl -s -X POST http://localhost:3000/maintenance_records -H "Content-Type: application/json" -d '{"maintenance_record":{"description":"Future","performed_at":"2027-01-01T12:00:00Z","equipment_id":2}}'
{
  "errors": [
    "Performed at cannot be in the future"
  ]
}

--- Rule 3: Category name minimum length (3 chars) ---
>> curl -s -X POST http://localhost:3000/categories -H "Content-Type: application/json" -d '{"category":{"name":"AB"}}'
{
  "errors": [
    "Name is too short (minimum is 3 characters)"
  ]
}

--- Rule 4: Equipment name must contain a letter ---
>> curl -s -X POST http://localhost:3000/equipment -H "Content-Type: application/json" -d '{"equipment":{"name":"123","serial_number":"NUM-001","status":"available","category_id":3}}'
{
  "errors": [
    "Name must contain at least one letter"
  ]
}

--- Rule 4b: Equipment name minimum length (3 chars) ---
>> curl -s -X POST http://localhost:3000/equipment -H "Content-Type: application/json" -d '{"equipment":{"name":"AB","serial_number":"SHT-001","status":"available","category_id":3}}'
{
  "errors": [
    "Name is too short (minimum is 3 characters)"
  ]
}

----------------------------------------------------------------------
TASK 7: Edge Cases (All 10)
----------------------------------------------------------------------

EC1: Create equipment with category_id that does not exist (expect 422)
>> curl -s -X POST http://localhost:3000/equipment -H "Content-Type: application/json" -d '{"equipment":{"name":"Edge Test","serial_number":"EDG-001","status":"available","category_id":999}}'
{
  "errors": [
    "Category must exist"
  ]
}

EC2: Create equipment with duplicate serial number (expect 422)
>> curl -s -X POST http://localhost:3000/equipment -H "Content-Type: application/json" -d '{"equipment":{"name":"Duplicate Serial","serial_number":"LAP-001","status":"available","category_id":3}}'
{
  "errors": [
    "Serial number has already been taken"
  ]
}

EC3: Create equipment with invalid status 'broken' (expect 422)
>> curl -s -X POST http://localhost:3000/equipment -H "Content-Type: application/json" -d '{"equipment":{"name":"Broken Status","serial_number":"STS-001","status":"broken","category_id":3}}'
{
  "errors": [
    "Status is not included in the list"
  ]
}

EC4: Create category with duplicate name (expect 422)
>> curl -s -X POST http://localhost:3000/categories -H "Content-Type: application/json" -d '{"category":{"name":"Computing"}}'
{
  "errors": [
    "Name has already been taken"
  ]
}

EC5: Create maintenance record with equipment_id that does not exist (expect 422)
>> curl -s -X POST http://localhost:3000/maintenance_records -H "Content-Type: application/json" -d '{"maintenance_record":{"description":"Bad equipment edge","performed_at":"2026-05-26T12:00:00Z","equipment_id":999}}'
{
  "errors": [
    "Equipment must exist"
  ]
}

EC6: DELETE /categories/:id when equipment still belongs to it (expect 409)
>> curl -s -X DELETE http://localhost:3000/categories/3
{
  "error": "Cannot delete category. 3 equipment items still belong to it."
}

EC7: GET /categories/999 (expect 404)
>> curl -s http://localhost:3000/categories/999
{
  "error": "Category not found"
}

EC8: GET /equipment/999 (expect 404)
>> curl -s http://localhost:3000/equipment/999
{
  "error": "Equipment not found"
}

EC9: PATCH /categories/999 (expect 404)
>> curl -s -X PATCH http://localhost:3000/categories/999 -H "Content-Type: application/json" -d '{"category":{"name":"Nope"}}'
{
  "error": "Category not found"
}

EC10: POST maintenance record with future performed_at (expect 422)
>> curl -s -X POST http://localhost:3000/maintenance_records -H "Content-Type: application/json" -d '{"maintenance_record":{"description":"Future edge","performed_at":"2030-06-01T12:00:00Z","equipment_id":2}}'
{
  "errors": [
    "Performed at cannot be in the future"
  ]
}
